-- ============================================================ -- Roster SaaS — v2 migration (run once, AFTER schema.sql) -- Adds: role types, in-app notifications, employee-initiated -- shift cancellation, and employee-to-employee shift swaps. -- Purely additive — safe to run on a database that already has -- companies/employees/shifts in it. -- ============================================================ -- ---------- Role types ---------- create table if not exists roles ( id uuid primary key default gen_random_uuid(), company_id uuid not null references companies(id) on delete cascade, name text not null, created_at timestamptz not null default now(), unique (company_id, name) ); -- an employee can hold more than one role (e.g. both Waiter and Bartender) create table if not exists employee_roles ( employee_id uuid not null references profiles(id) on delete cascade, role_id uuid not null references roles(id) on delete cascade, primary key (employee_id, role_id) ); -- a shift slot can require a specific role (nullable = any employee is fine) alter table shift_slots add column if not exists role_id uuid references roles(id) on delete set null; create index if not exists idx_roles_company on roles(company_id); create index if not exists idx_employee_roles_employee on employee_roles(employee_id); create index if not exists idx_employee_roles_role on employee_roles(role_id); create index if not exists idx_slots_role on shift_slots(role_id); alter table roles enable row level security; alter table employee_roles enable row level security; drop policy if exists roles_select on roles; create policy roles_select on roles for select using (company_id = my_company_id()); drop policy if exists roles_insert on roles; create policy roles_insert on roles for insert with check (my_role() = 'admin' and company_id = my_company_id()); drop policy if exists roles_delete on roles; create policy roles_delete on roles for delete using (my_role() = 'admin' and company_id = my_company_id()); drop policy if exists employee_roles_select on employee_roles; create policy employee_roles_select on employee_roles for select using ( employee_id in (select id from profiles where company_id = my_company_id()) ); drop policy if exists employee_roles_insert on employee_roles; create policy employee_roles_insert on employee_roles for insert with check ( my_role() = 'admin' and employee_id in (select id from profiles where company_id = my_company_id()) and role_id in (select id from roles where company_id = my_company_id()) ); drop policy if exists employee_roles_delete on employee_roles; create policy employee_roles_delete on employee_roles for delete using ( my_role() = 'admin' and employee_id in (select id from profiles where company_id = my_company_id()) ); -- ---------- In-app notifications ---------- create table if not exists notifications ( id uuid primary key default gen_random_uuid(), company_id uuid not null references companies(id) on delete cascade, recipient_id uuid not null references profiles(id) on delete cascade, actor_id uuid references profiles(id) on delete set null, type text not null check (type in ('shift_cancelled','swap_requested','swap_accepted','swap_declined')), message text not null, slot_id uuid references shift_slots(id) on delete set null, is_read boolean not null default false, created_at timestamptz not null default now() ); create index if not exists idx_notifications_recipient on notifications(recipient_id, is_read); alter table notifications enable row level security; drop policy if exists notifications_select on notifications; create policy notifications_select on notifications for select using (recipient_id = auth.uid()); drop policy if exists notifications_update on notifications; create policy notifications_update on notifications for update using (recipient_id = auth.uid()); -- No insert policy: notifications are only ever written by the security-definer -- functions below, which run with elevated privilege and bypass RLS for the insert. -- ---------- Shift swap requests ---------- create table if not exists swap_requests ( id uuid primary key default gen_random_uuid(), company_id uuid not null references companies(id) on delete cascade, assignment_id uuid not null references shift_assignments(id) on delete cascade, requested_by uuid not null references profiles(id) on delete cascade, requested_to uuid not null references profiles(id) on delete cascade, direction text not null check (direction in ('offer','request')), status text not null default 'pending' check (status in ('pending','accepted','declined','cancelled')), note text, created_at timestamptz not null default now(), responded_at timestamptz ); create index if not exists idx_swap_requested_to on swap_requests(requested_to, status); create index if not exists idx_swap_requested_by on swap_requests(requested_by, status); alter table swap_requests enable row level security; drop policy if exists swap_select on swap_requests; create policy swap_select on swap_requests for select using ( requested_by = auth.uid() or requested_to = auth.uid() or (my_role() = 'admin' and company_id = my_company_id()) ); -- No insert/update policy: all writes go through the functions below, which -- validate ownership and direction before touching anything. -- ---------- Functions ---------- -- Employee cancels/declines a shift they're currently assigned to. Frees the -- seat back to "open" and notifies the company's admin. create or replace function employee_cancel_shift(p_assignment_id uuid, p_reason text default null) returns void language plpgsql security definer set search_path = public as $$ declare v_employee_id uuid; v_slot record; v_employee_name text; begin select employee_id into v_employee_id from shift_assignments where id = p_assignment_id; if v_employee_id is null or v_employee_id <> auth.uid() then raise exception 'You can only cancel your own shifts'; end if; select ss.date, ss.start_time, ss.end_time, ss.id as slot_id, ss.company_id, c.owner_id into v_slot from shift_assignments sa join shift_slots ss on ss.id = sa.slot_id join companies c on c.id = ss.company_id where sa.id = p_assignment_id; select full_name into v_employee_name from profiles where id = auth.uid(); delete from shift_assignments where id = p_assignment_id; insert into notifications(company_id, recipient_id, actor_id, type, message, slot_id) values ( v_slot.company_id, v_slot.owner_id, auth.uid(), 'shift_cancelled', v_employee_name || ' can no longer work ' || to_char(v_slot.date, 'Dy DD Mon') || ' ' || to_char(v_slot.start_time, 'HH24:MI') || '\u2013' || to_char(v_slot.end_time, 'HH24:MI') || coalesce(' \u2014 ' || nullif(p_reason, ''), ''), v_slot.slot_id ); end; $$; grant execute on function employee_cancel_shift(uuid, text) to authenticated; -- Create a swap request. direction='offer' = requested_by is giving away -- their own shift to requested_to. direction='request' = requested_by wants -- to take requested_to's shift. create or replace function create_swap_request(p_assignment_id uuid, p_requested_to uuid, p_direction text, p_note text default null) returns uuid language plpgsql security definer set search_path = public as $$ declare v_owner uuid; v_company_id uuid; v_new_id uuid; v_requester_name text; v_slot record; begin if p_direction not in ('offer','request') then raise exception 'Invalid direction'; end if; if p_requested_to = auth.uid() then raise exception 'Pick a different teammate'; end if; select sa.employee_id, ss.company_id into v_owner, v_company_id from shift_assignments sa join shift_slots ss on ss.id = sa.slot_id where sa.id = p_assignment_id; if v_owner is null then raise exception 'Shift not found'; end if; if v_company_id <> my_company_id() then raise exception 'Not your company'; end if; if (select company_id from profiles where id = p_requested_to) <> my_company_id() then raise exception 'That person is not on your team'; end if; if p_direction = 'offer' then if v_owner <> auth.uid() then raise exception 'You can only offer away your own shift'; end if; else if v_owner <> p_requested_to then raise exception 'That is not their shift'; end if; end if; insert into swap_requests(company_id, assignment_id, requested_by, requested_to, direction, note) values (v_company_id, p_assignment_id, auth.uid(), p_requested_to, p_direction, nullif(p_note,'')) returning id into v_new_id; select full_name into v_requester_name from profiles where id = auth.uid(); select ss.date, ss.start_time, ss.end_time, ss.id as slot_id into v_slot from shift_slots ss join shift_assignments sa on sa.slot_id = ss.id where sa.id = p_assignment_id; insert into notifications(company_id, recipient_id, actor_id, type, message, slot_id) values ( v_company_id, p_requested_to, auth.uid(), 'swap_requested', v_requester_name || (case when p_direction = 'offer' then ' wants to give you their shift on ' else ' wants to take your shift on ' end) || to_char(v_slot.date, 'Dy DD Mon') || ' ' || to_char(v_slot.start_time,'HH24:MI') || '\u2013' || to_char(v_slot.end_time,'HH24:MI'), v_slot.slot_id ); return v_new_id; end; $$; grant execute on function create_swap_request(uuid, uuid, text, text) to authenticated; -- Recipient accepts or declines a swap request aimed at them. create or replace function respond_to_swap_request(p_request_id uuid, p_accept boolean) returns void language plpgsql security definer set search_path = public as $$ declare v_req record; v_current_owner uuid; v_slot record; v_responder_name text; begin select * into v_req from swap_requests where id = p_request_id; if v_req is null then raise exception 'Request not found'; end if; if v_req.requested_to <> auth.uid() then raise exception 'This request is not yours to answer'; end if; if v_req.status <> 'pending' then raise exception 'This request has already been resolved'; end if; select full_name into v_responder_name from profiles where id = auth.uid(); if not p_accept then update swap_requests set status = 'declined', responded_at = now() where id = p_request_id; insert into notifications(company_id, recipient_id, actor_id, type, message) values (v_req.company_id, v_req.requested_by, auth.uid(), 'swap_declined', v_responder_name || ' declined your shift swap request'); return; end if; select employee_id into v_current_owner from shift_assignments where id = v_req.assignment_id; if v_current_owner is null then raise exception 'That shift no longer exists'; end if; if v_req.direction = 'offer' then if v_current_owner <> v_req.requested_by then raise exception 'That shift has already changed hands'; end if; update shift_assignments set employee_id = v_req.requested_to, assigned_by = 'manual' where id = v_req.assignment_id; else if v_current_owner <> v_req.requested_to then raise exception 'That shift has already changed hands'; end if; update shift_assignments set employee_id = v_req.requested_by, assigned_by = 'manual' where id = v_req.assignment_id; end if; update swap_requests set status = 'accepted', responded_at = now() where id = p_request_id; select ss.date, ss.start_time, ss.end_time, ss.id as slot_id into v_slot from shift_slots ss join shift_assignments sa on sa.slot_id = ss.id where sa.id = v_req.assignment_id; insert into notifications(company_id, recipient_id, actor_id, type, message, slot_id) values ( v_req.company_id, v_req.requested_by, auth.uid(), 'swap_accepted', v_responder_name || ' accepted your shift swap for ' || to_char(v_slot.date,'Dy DD Mon') || ' ' || to_char(v_slot.start_time,'HH24:MI') || '\u2013' || to_char(v_slot.end_time,'HH24:MI'), v_slot.slot_id ); end; $$; grant execute on function respond_to_swap_request(uuid, boolean) to authenticated; -- Requester cancels their own still-pending swap request. create or replace function cancel_swap_request(p_request_id uuid) returns void language plpgsql security definer set search_path = public as $$ begin update swap_requests set status = 'cancelled', responded_at = now() where id = p_request_id and requested_by = auth.uid() and status = 'pending'; if not found then raise exception 'Nothing to cancel'; end if; end; $$; grant execute on function cancel_swap_request(uuid) to authenticated;
